module mux_41(input I0,I1,I2,I3,en,input [3:2] sel, output y);
  assign y = (en==0) ? 1'b0 : (I0 & ~sel[3] & ~sel[2]) | (I1 & ~sel[3] & sel[2]) |(I2 & sel[3] & ~sel[2]) |(I3 	  & sel[3] & sel[2]);
endmodule
