module decoder(
  input [2:0] D,
  output reg [7:0] A
);

always@(*) begin
  case(D)
    3'b000:A=8'b10000000;
    3'b001:A=8'b01000000;
    3'b010:A=8'b00100000;
    3'b011:A=8'b00010000;
    3'b100:A=8'b00001000;
    3'b101:A=8'b00000100;
    3'b110:A=8'b00000010;
    3'b111:A=8'b00000000;
    default: A=8'b00000000;
  endcase
end

endmodule
