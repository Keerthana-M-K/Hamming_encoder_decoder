module encoder_ttb;
  reg [7:0] Y ;
  wire [2:0] A;
  encoder dut(
    .Y(Y),
    .A(A));
  
  initial  begin
    $monitor("Input Y= %b Output A2 = %b A1 = %b A0 = %b",Y,A[2],A[1],A[0]);
  Y=8'b00000001;
  #10 Y=8'b00000010;
  #10 Y=8'b00000100;
  #10 Y=8'b00001000;
  #10 Y=8'b00010000;
  #10 Y=8'b00100000;
  #10 Y=8'b01000000;
  #10 Y=8'b10000000;
  #10  $finish;
  end
endmodule
